cmake_minimum_required(VERSION 3.20)
project(tvrock_srt LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "TvRockDTV SRT requires Windows MSVC")
endif()
include(cmake/DetectMsvcIncludes.cmake)
add_subdirectory(cmake/dependencies)
add_library(tvrock_srt STATIC srt.cpp)
target_compile_features(tvrock_srt PRIVATE cxx_std_17)
target_link_libraries(tvrock_srt PRIVATE bondriver_srt_sdk)
install(TARGETS tvrock_srt srt_static mbedcrypto ARCHIVE DESTINATION lib)
install(FILES "${SRT_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME SRT.txt)
install(FILES "${MBEDTLS_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME MbedTLS.txt)
